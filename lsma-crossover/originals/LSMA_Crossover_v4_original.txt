//@version=4
study(title = "Least Squares Moving Average Crossover", shorttitle="LSMA Crossover", overlay=true, resolution="")
length = input(title="Length", type=input.integer, defval=21)
offset = input(title="Offset", type=input.integer, defval=0)
length2 = input(title="Trigger Length", type=input.integer, defval=4)
src = input(close, title="Source")
lsma = linreg(src, length, offset)
d = sma(lsma,length2)
lsmalong = linreg(src,200,0)
lsmaxlong = linreg(src,1000,0)
plot(lsma, linewidth=3, color=color.blue, title = "LSMA")
plot(d, linewidth=3, color=color.yellow, title = "Trigger")
plot(lsmalong, linewidth=3, color=color.white, title = "Long")
plot(lsmaxlong, linewidth=3, color=color.blue, title = "Extra Long")
